// contextual action button (handles the driver-assign popover inline)
function ActionBtn({ r, onAdvance, onAssign }) {
  const [open, setOpen] = useState(false);
  const step = nextStep(r.status);
  if (!step) return null;
  if (step.assign) {
    return (
      <span style={{ position: "relative" }}>
        <button onClick={() => setOpen((o) => !o)} style={primaryBtn}><LIcon name="user" size={15} />{step.label}</button>
        {open ? <DriverPicker onClose={() => setOpen(false)} onPick={(d) => { setOpen(false); onAssign(r.id, d); }} /> : null}
      </span>
    );
  }
  return (
    <button onClick={() => onAdvance(r.id, step.to)} style={step.primary ? primaryBtn : ghostBtn}>
      <LIcon name={step.icon} size={15} />{step.label}
    </button>
  );
}

const FILTERS = [
  { key: "all", label: "Sve", test: () => true },
  { key: "action", label: "Treba akcija", test: (r) => SMETA[r.status].kind === "action" },
  { key: "wait", label: "Čeka", test: (r) => SMETA[r.status].kind === "wait" },
  { key: "live", label: "U toku", test: (r) => SMETA[r.status].kind === "live" },
  { key: "done", label: "Završeno", test: (r) => ["done", "bad"].includes(SMETA[r.status].kind) },
];
function rankRow(r) {
  if (isLate(r)) return 0;
  const k = SMETA[r.status].kind;
  return k === "action" ? 1 : k === "wait" ? 2 : k === "live" ? 3 : 4;
}

const pct = (min) => `${((min - DAY_START) / DAY_SPAN) * 100}%`;
function laneOverlaps(items) {
  // mark blocks that overlap another in same lane
  const bad = new Set();
  for (let i = 0; i < items.length; i++)
    for (let j = i + 1; j < items.length; j++)
      if (t2m(items[i].from) < t2m(items[j].to) && t2m(items[j].from) < t2m(items[i].to)) { bad.add(items[i].id); bad.add(items[j].id); }
  return bad;
}

    const el = blockRef.current; if (!el) return;

    const vw = window.innerWidth, vh = window.innerHeight, m = 10;

    const roomBelow = vh - b.bottom, roomAbove = b.top;
    const flip = roomBelow < POP_H + 12 && roomAbove > roomBelow;
    let top = flip ? b.top - 8 : b.bottom + 8;
    if (!flip) top = Math.min(top, vh - POP_H - m);          // clamp so it never clips bottom

function Timeline({ reqs, onAdvance, onAssign, onOpenDetail }) {
  const hours = [];
  for (let h = DAY_START / 60; h <= DAY_END / 60; h++) hours.push(h);
  const lanes = [{ key: "_un", label: "Nedodeljeno", driver: null }, ...DRIVERS.map((d) => ({ key: d, label: d, driver: d }))];
  const colW = 78; // px per hour

      <div style={{ overflowX: "auto" }}>
        <div style={{ minWidth: trackW + 150, width: "100%" }}>

            <div style={{ position: "relative", flex: 1, minWidth: trackW, minHeight: 46 }}>

                <span key={h} style={{ position: "absolute", left: pct(h * 60), transform: "translateX(-50%)", top: 11, fontFamily: "var(--font-mono)", fontSize: 10.5, color: "var(--text-faint)" }}>{pad(h)}:00</span>
              ))}
              {/* now line label — own row below the hour numbers so it doesn't overlap */}
              <span style={{ position: "absolute", left: pct(NOW_MIN), transform: "translateX(-50%)", top: 27, display: "inline-flex", alignItems: "center", gap: 4, fontFamily: "var(--font-mono)", fontSize: 9.5, fontWeight: 600, color: "#10231a", background: "var(--mint-grad)", padding: "2px 7px", borderRadius: 999, whiteSpace: "nowrap" }}>SAD {m2t(NOW_MIN)}</span>
            </div>
          </div>
          {/* lanes */}
          {lanes.map((lane, li) => {
            const items = reqs.filter((r) => lane.driver ? r.driver === lane.driver : !r.driver).sort((a, b) => t2m(a.from) - t2m(b.from));

                </div>

          <div style={{ fontFamily: "var(--font-display)", fontSize: 14.5, fontWeight: 600, whiteSpace: "nowrap", overflow: "hidden", textOverflow: "ellipsis" }}>{r.make}</div>
          <div style={{ fontSize: 11.5, color: "var(--text-dim)", whiteSpace: "nowrap", overflow: "hidden", textOverflow: "ellipsis" }}>{r.client}</div>
        </div>
      </div>
      <div style={{ display: "flex", alignItems: "center", gap: 10, flexWrap: "wrap" }}>
        <Meta icon="clock" warn={late}>{r.from}–{r.to}</Meta>
        <span style={{ fontFamily: "var(--font-mono)", fontSize: 9.5, color: "var(--text-faint)", border: "1px solid var(--glass-line)", padding: "1px 6px", borderRadius: 5 }}>{r.plate}</span>
      </div>
      <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", gap: 8 }}>
        <DriverChip name={r.driver} />
        {step ? (
          <button onClick={(e) => { e.stopPropagation(); step.assign ? onOpenDetail(r.id) : onAdvance(r.id, step.to); }}
            title={step.label}
            style={{ display: "inline-flex", alignItems: "center", gap: 5, fontFamily: "var(--font-body)", fontWeight: 600, fontSize: 11.5, padding: "6px 10px", borderRadius: 9, border: "none", cursor: "pointer", background: step.primary ? "var(--brass-grad)" : "rgba(255,255,255,0.06)", color: step.primary ? "#1a130a" : "var(--text)", whiteSpace: "nowrap", flexShrink: 0 }}>
            <LIcon name={step.icon} size={13} />{step.label}
          </button>
        ) : null}
      </div>
    </div>
  );
}
function Kanban({ reqs, onAdvance, onAssign, onOpenDetail, flashId }) {
  return (
    <div className="rd-fade" style={{ overflowX: "auto", paddingBottom: 8 }}>
      <div style={{ display: "flex", gap: 14, minWidth: "min-content" }}>
        {KCOLS.map((col, ci) => {
          const items = reqs.filter((r) => col.statuses.includes(r.status)).sort((a, b) => t2m(a.from) - t2m(b.from));
          return (
            <div key={col.key} className="rd-rise" style={{ width: 264, flexShrink: 0, display: "flex", flexDirection: "column", animationDelay: `${ci * 0.05}s` }}>
              {/* column header */}
              <div style={{ display: "flex", alignItems: "center", gap: 9, padding: "0 4px 12px" }}>
                <span style={{ width: 8, height: 8, borderRadius: "50%", background: col.accent, boxShadow: `0 0 8px ${col.accent}` }} />
                <span style={{ fontFamily: "var(--font-display)", fontSize: 14.5, fontWeight: 600 }}>{col.label}</span>
                <span style={{ fontFamily: "var(--font-mono)", fontSize: 11, color: "var(--text-faint)", background: "rgba(255,255,255,0.05)", padding: "2px 8px", borderRadius: 999 }}>{items.length}</span>
              </div>
              {/* cards */}
              <div style={{ display: "flex", flexDirection: "column", gap: 10, background: "rgba(255,255,255,0.018)", border: "1px solid var(--glass-line)", borderRadius: 16, padding: 10, minHeight: 120, flex: 1 }}>
                {items.length === 0 ? (
                  <div style={{ padding: "26px 8px", textAlign: "center", fontSize: 12, color: "var(--text-faint)" }}>—</div>
                ) : items.map((r) => (
                  <KanbanCard key={r.id} r={r} onAdvance={onAdvance} onAssign={onAssign} onOpenDetail={onOpenDetail} flash={flashId === r.id} />
                ))}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}

// view switcher (segmented)
function Switcher({ view, setView }) {
  const opts = [{ key: "timeline", label: "Raspored", icon: "timeline" }, { key: "list", label: "Lista", icon: "list" }, { key: "kanban", label: "Kanban", icon: "kanban" }];
  return (
    <div style={{ display: "inline-flex", padding: 4, gap: 4, borderRadius: 13, background: "rgba(8,14,11,0.5)", border: "1px solid var(--glass-line)" }}>
      {opts.map((o) => {
        const on = view === o.key;
        return (
          <button key={o.key} onClick={() => setView(o.key)} style={{ display: "inline-flex", alignItems: "center", gap: 8, fontFamily: "var(--font-body)", fontSize: 13.5, fontWeight: on ? 600 : 500, padding: "8px 16px", borderRadius: 10, cursor: "pointer", border: "none", color: on ? "#10231a" : "var(--text-dim)", background: on ? "var(--mint-grad)" : "transparent", transition: "all .15s" }}>
            <LIcon name={o.icon} size={15} />{o.label}
          </button>
        );
      })}
    </div>
  );
}

// ── Root ─────────────────────────────────────────────────────
function DispatcherBoardApp() {
  const [reqs, setReqs] = useState(SEED);
  const [view, setView] = useState("timeline");
  const [flashId, setFlashId] = useState(null);
  const [detailId, setDetailId] = useState(null);
  const detailReq = reqs.find((r) => r.id === detailId) || null;

  const advance = (id, to) => {
    setReqs((rs) => rs.map((r) => r.id === id ? { ...r, status: to } : r));
    setFlashId(id); setTimeout(() => setFlashId(null), 1100);
  };
  const assign = (id, driver) => {
    setReqs((rs) => rs.map((r) => r.id === id ? { ...r, driver, status: r.status === "CONFIRMED" ? "DRIVER_ASSIGNED" : r.status } : r));
    setFlashId(id); setTimeout(() => setFlashId(null), 1100);
  };

  return (
    <div data-role="dispatcher" style={{ minHeight: "100vh" }}>
      <Header />
      <div style={page}>
        <div className="rd-rise" style={{ display: "flex", alignItems: "flex-end", justifyContent: "space-between", flexWrap: "wrap", gap: 14, marginBottom: 22 }}>
          <div>
            <h1 style={{ fontFamily: "var(--font-display)", fontSize: 30, fontWeight: 700, letterSpacing: "-.8px", margin: "0 0 5px" }}>Dispečerska tabla</h1>
            <p style={{ fontSize: 14, color: "var(--text-dim)", margin: 0 }}>{view === "timeline" ? "Raspored dana po vozaču — klik na blok otvara akcije i detalje." : view === "kanban" ? "Zahtevi grupisani po statusu — klik na karticu otvara detalje." : "Šta traži pažnju, sortirano po hitnosti."}</p>
          </div>
          <Switcher view={view} setView={setView} />
        </div>
        <Kpis reqs={reqs} />
        {view === "timeline"
          ? <Timeline reqs={reqs} onAdvance={advance} onAssign={assign} onOpenDetail={setDetailId} />
          : view === "kanban"
          ? <Kanban reqs={reqs} onAdvance={advance} onAssign={assign} onOpenDetail={setDetailId} flashId={flashId} />
          : <Worklist reqs={reqs} onAdvance={advance} onAssign={assign} onOpenDetail={setDetailId} flashId={flashId} />}
      </div>
      {detailReq ? <DetailDrawer r={detailReq} onClose={() => setDetailId(null)} onAdvance={(id, to) => { advance(id, to); }} onAssign={(id, d) => { assign(id, d); }} /> : null}
    </div>
  );
}

window.DispatcherBoardApp = DispatcherBoardApp;
